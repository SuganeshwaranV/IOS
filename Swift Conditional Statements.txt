Swift Conditional Statements

if statement:

let age = 21

if age >= 18 {
    print("You are eligible")
}

Output:

You are eligible

let age = 16

if age >= 18 {
    print("You are eligible")
}

print("Program ended")

Output:

Program ended


if-else statement:

let age = 16

if age >= 18 {
    print("Adult")
} else {
    print("Minor")
}

Output:

Minor

let isLoggedIn = false

if isLoggedIn {
    print("Welcome to the app")
} else {
    print("Please log in")
}

Output:

Please log in

if-else if-else statement:

let marks = 82

if marks >= 90 {
    print("Grade A")
} else if marks >= 75 {
    print("Grade B")
} else if marks >= 50 {
    print("Grade C")
} else {
    print("Fail")
}

Output:

Grade B

Nested if statements

let isLoggedIn = true
let isPremium = true

if isLoggedIn {
    print("Login successful")

    if isPremium {
        print("Premium features unlocked")
    } else {
        print("Free account")
    }

} else {
    print("Please log in")
}

Output:

Login successful
Premium features unlocked
